import pygame import math import random import sys # ========================================== # 1. 게임 기본 설정 (전체 환경) # ========================================== pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("미니의 통합 게임 박스 - 종합 선물 세트") clock = pygame.time.Clock() FPS = 60 # 색상 정의 WHITE = (255, 255, 255) BLACK = (0, 0, 0) RED = (230, 50, 50) GREEN = (50, 200, 50) BLUE = (50, 100, 230) GRAY = (200, 200, 200) YELLOW = (240, 200, 50) DARK_GRAY = (40, 40, 50) # 폰트 설정 try: font = pygame.font.SysFont("malgungothic", 20) title_font = pygame.font.SysFont("malgungothic", 36) except: font = pygame.font.Font(None, 24) title_font = pygame.font.Font(None, 48) # 전체 게임 상태 (0: 메인메뉴, 1: 타워 디펜스, 2: 삼매치, 3: 스마트폰 슈팅) game_state = 0 # ========================================== # 2. [게임 1] 타워 디펜스 클래스 모음 # ========================================== WAYPOINTS = [(0, 300), (200, 300), (200, 150), (500, 150), (500, 450), (800, 450)] class Enemy: def __init__(self, enemy_type): self.type = enemy_type self.x, self.y = WAYPOINTS[0] self.target_idx = 1 if enemy_type == 1: self.hp = 100 self.max_hp = 100 self.speed = 2.0 self.color = RED self.size = 12 self.reward = 15 elif enemy_type == 2: self.hp = 60 self.max_hp = 60 self.speed = 3.8 self.color = YELLOW self.size = 9 self.reward = 20 elif enemy_type == 3: self.hp = 250 self.max_hp = 250 self.speed = 1.2 self.color = (130, 50, 200) self.size = 16 self.reward = 35 def move(self): if self.target_idx < len(WAYPOINTS): tx, ty = WAYPOINTS[self.target_idx] dx, dy = tx - self.x, ty - self.y dist = math.hypot(dx, dy) if dist < self.speed: self.x, self.y = tx, ty self.target_idx += 1 else: self.x += (dx / dist) * self.speed self.y += (dy / dist) * self.speed return self.target_idx >= len(WAYPOINTS) def draw(self, surface): pygame.draw.circle(surface, self.color, (int(self.x), int(self.y)), self.size) hp_bar_width = 20 hp_ratio = max(0, self.hp / self.max_hp) pygame.draw.rect(surface, RED, (self.x - 10, self.y - self.size - 6, hp_bar_width, 4)) pygame.draw.rect(surface, GREEN, (self.x - 10, self.y - self.size - 6, hp_bar_width * hp_ratio, 4)) class Tower: def __init__(self, x, y): self.x = x self.y = y self.range = 120 self.damage = 25 self.cooldown = 0 self.fire_rate = 30 def attack(self, enemies, bullets): if self.cooldown > 0: self.cooldown -= 1 return for enemy in enemies: dist = math.hypot(enemy.x - self.x, enemy.y - self.y) if dist <= self.range: bullets.append(Bullet(self.x, self.y, enemy, self.damage)) self.cooldown = self.fire_rate break def draw(self, surface): pygame.draw.circle(surface, BLUE, (self.x, self.y), 15) pygame.draw.circle(surface, BLACK, (self.x, self.y), 15, 2) class Bullet: def __init__(self, x, y, target, damage): self.x = x self.y = y self.target = target self.damage = damage self.speed = 8.0 self.hit = False def move(self): if self.target not in defense_game.enemies: self.hit = True return dx = self.target.x - self.x dy = self.target.y - self.y dist = math.hypot(dx, dy) if dist < self.speed: self.target.hp -= self.damage self.hit = True else: self.x += (dx / dist) * self.speed self.y += (dy / dist) * self.speed def draw(self, surface): pygame.draw.circle(surface, BLACK, (int(self.x), int(self.y)), 4) # ========================================== # 3. 디펜스 게임 메인 관리자 # ========================================== class DefenseGame: def __init__(self): self.reset() def reset(self): self.money = 150 self.life = 10 self.wave = 1 self.towers = [] self.enemies = [] self.bullets = [] self.spawn_timer = 0 self.enemies_spawned = 0 self.max_enemies_per_wave = 10 self.tower_cost = 50 def update(self): self.spawn_timer += 1 if self.spawn_timer > 60 and self.enemies_spawned < self.max_enemies_per_wave: self.spawn_timer = 0 self.enemies_spawned += 1 if self.wave == 1: e_type = 1 elif self.wave == 2: e_type = 2 if self.enemies_spawned % 2 == 0 else 1 else: r = self.enemies_spawned % 3 e_type = 3 if r == 0 else (2 if r == 1 else 1) self.enemies.append(Enemy(e_type)) if self.enemies_spawned >= self.max_enemies_per_wave and len(self.enemies) == 0: self.wave += 1 self.enemies_spawned = 0 self.max_enemies_per_wave += 5 self.money += 50 for enemy in self.enemies[:]: if enemy.move(): self.life -= 1 self.enemies.remove(enemy) elif enemy.hp <= 0: self.money += enemy.reward self.enemies.remove(enemy) for tower in self.towers: tower.attack(self.enemies, self.bullets) for bullet in self.bullets[:]: bullet.move() if bullet.hit: self.bullets.remove(bullet) def draw(self, surface): surface.fill((230, 240, 230)) if len(WAYPOINTS) > 1: pygame.draw.lines(surface, GRAY, False, WAYPOINTS, 40) for tower in self.towers: tower.draw(surface) for enemy in self.enemies: enemy.draw(surface) for bullet in self.bullets: bullet.draw(surface) info_text = f"목숨: {self.life} | 돈: ${self.money} | 웨이브: {self.wave} | 타워 설치 비용: ${self.tower_cost}" txt_surface = font.render(info_text, True, BLACK) surface.blit(txt_surface, (15, 15)) btn_txt = font.render("[ESC] 메인 메뉴로", True, RED) surface.blit(btn_txt, (WIDTH - 150, 15)) if self.life <= 0: over_txt = title_font.render("GAME OVER! (R: 재시작)", True, RED) surface.blit(over_txt, (WIDTH//2 - 180, HEIGHT//2)) def handle_click(self, pos): if self.money >= self.tower_cost: px, py = pos on_path = False for i in range(len(WAYPOINTS)-1): p1, p2 = WAYPOINTS[i], WAYPOINTS[i+1] dist = math.hypot((p1[0]+p2[0])/2 - px, (p1[1]+p2[1])/2 - py) if dist < 30: on_path = True break if not on_path and py > 50: self.towers.append(Tower(px, py)) self.money -= self.tower_cost # 디펜스 게임 객체 생성 defense_game = DefenseGame() # ========================================== # 4. [게임 3] 스마트폰 슈팅 게임 클래스 모음 # ========================================== class Target: def __init__(self): self.reset() def reset(self): self.x = random.randint(80, WIDTH - 80) self.y = random.randint(100, HEIGHT - 150) self.size = random.randint(15, 25) self.max_size = random.randint(55, 75) self.grow_speed = random.uniform(0.3, 0.7) self.type = random.choices([1, 2, 3], weights=[60, 25, 15])[0] if self.type == 1: self.color = RED self.score_val = 100 elif self.type == 2: self.color = YELLOW self.grow_speed *= 1.5 self.score_val = 200 elif self.type == 3: self.color = GREEN self.score_val = 300 self.is_active = True def update(self): if self.is_active: self.size += self.grow_speed if self.size >= self.max_size: self.is_active = False return True return False def draw(self, surface): if self.is_active: int_size = int(self.size) pygame.draw.circle(surface, self.color, (int(self.x), int(self.y)), int_size) pygame.draw.circle(surface, WHITE, (int(self.x), int(self.y)), int(max(1, int_size * 0.6))) pygame.draw.circle(surface, self.color, (int(self.x), int(self.y)), int(max(1, int_size * 0.3))) def is_hit(self, pos): mx, my = pos dist = math.hypot(self.x - mx, self.y - my) return dist <= self.size class HitEffect: def __init__(self, x, y): self.x = x self.y = y self.timer = 10 def draw(self, surface): if self.timer > 0: pygame.draw.circle(surface, YELLOW, (self.x, self.y), 12, 2) pygame.draw.line(surface, YELLOW, (self.x - 15, self.y), (self.x + 15, self.y), 2) pygame.draw.line(surface, YELLOW, (self.x, self.y - 15), (self.x, self.y + 15), 2) self.timer -= 1 class MobileFPSGame: def __init__(self): self.reset() def reset(self): self.score = 0 self.health = 5 self.ammo = 7 self.max_ammo = 7 self.is_reloading = False self.reload_timer = 0 self.targets = [Target() for _ in range(4)] self.effects = [] self.game_over = False def reload(self): if not self.is_reloading and self.ammo < self.max_ammo: self.is_reloading = True self.reload_timer = 45 def update(self): if self.game_over: return if self.is_reloading: self.reload_timer -= 1 if self.reload_timer <= 0: self.ammo = self.max_ammo self.is_reloading = False for target in self.targets: missed = target.update() if missed: self.health -= 1 target.reset() if self.health <= 0: self.game_over = True def shoot(self, pos): if self.game_over: return if self.is_reloading: return if self.ammo <= 0: self.reload() return self.ammo -= 1 self.effects.append(HitEffect(pos[0], pos[1])) for target in self.targets: if target.is_active and target.is_hit(pos): self.score += target.score_val target.reset() break if self.ammo == 0: self.reload() def draw(self, surface): surface.fill(DARK_GRAY) pygame.draw.polygon(surface, (25, 25, 30), [(0, HEIGHT), (WIDTH, HEIGHT), (WIDTH*0.8, HEIGHT*0.6), (WIDTH*0.2, HEIGHT*0.6)]) pygame.draw.line(surface, (70, 70, 80), (0, HEIGHT*0.6), (WIDTH, HEIGHT*0.6), 2) for target in self.targets: target.draw(surface) for effect in self.effects[:]: effect.draw(surface) if effect.timer <= 0: self.effects.remove(effect) # 조준경 mx, my = pygame.mouse.get_pos() pygame.draw.circle(surface, GREEN, (mx, my), 18, 2) pygame.draw.circle(surface, RED, (mx, my), 3) pygame.draw.line(surface, GREEN, (mx - 25, my), (mx + 25, my), 1) pygame.draw.line(surface, GREEN, (mx, my - 25), (mx, my + 25), 1) # UI score_txt = font.render(f"SCORE: {self.score}", True, WHITE) health_txt = font.render(f"HP: {'❤️' * self.health}", True, RED) surface.blit(score_txt, (20, 20)) surface.blit(health_txt, (20, 55)) btn_txt = font.render("[ESC] 메인 메뉴로", True, RED) surface.blit(btn_txt, (WIDTH - 150, 15)) if self.is_reloading: ammo_str = "RELOADING..." ammo_color = YELLOW else: ammo_str = f"AMMO: {'▮ ' * self.ammo}" ammo_color = GREEN ammo_txt = font.render(ammo_str, True, ammo_color) surface.blit(ammo_txt, (WIDTH - 220, HEIGHT - 50)) reload_btn_txt = font.render("[R] 키: 장전", True, WHITE) surface.blit(reload_btn_txt, (WIDTH - 220, HEIGHT - 85)) if self.game_over: s = pygame.Surface((WIDTH, HEIGHT)) s.set_alpha(200) s.fill(BLACK) surface.blit(s, (0, 0)) over_txt = title_font.render("MISSION FAILED", True, RED) final_score = font.render(f"최종 점수: {self.score}", True, WHITE) restart_txt = font.render("클릭 / [R] 키를 눌러 다시 시작", True, YELLOW) surface.blit(over_txt, (WIDTH//2 - over_txt.get_width()//2, HEIGHT//2 - 80)) surface.blit(final_score, (WIDTH//2 - final_score.get_width()//2, HEIGHT//2)) surface.blit(restart_txt, (WIDTH//2 - restart_txt.get_width()//2, HEIGHT//2 + 60)) # 슈팅 게임 객체 생성 fps_game = MobileFPSGame() # ========================================== # 5. 메인 메뉴 및 전체 게임 루프 # ========================================== def draw_main_menu(): screen.fill((240, 240, 245)) pygame.mouse.set_visible(True) # 메인 메뉴에서는 마우스 커서 켜기 title = title_font.render("🎮 미니의 종합 게임 박스", True, BLACK) screen.blit(title, (WIDTH//2 - title.get_width()//2, 80)) # 1번 게임 버튼 (타워 디펜스) pygame.draw.rect(screen, BLUE, (250, 180, 300, 55), border_radius=10) btn1_txt = font.render("1. 타워 디펜스 게임 시작", True, WHITE) screen.blit(btn1_txt, (WIDTH//2 - btn1_txt.get_width()//2, 195)) # 2번 게임 버튼 (삼매치) pygame.draw.rect(screen, GREEN, (250, 260, 300, 55), border_radius=10) btn2_txt = font.render("2. 삼매치 게임 (오리지널 연동)", True, WHITE) screen.blit(btn2_txt, (WIDTH//2 - btn2_txt.get_width()//2, 275)) # 3번 게임 버튼 (스마트폰 슈팅 게임) pygame.draw.rect(screen, RED, (250, 340, 300, 55), border_radius=10) btn3_txt = font.render("3. 스마트폰 FPS 슈팅 게임", True, WHITE) screen.blit(btn3_txt, (WIDTH//2 - btn3_txt.get_width()//2, 355)) guide = font.render("원하는 게임의 번호 버튼을 클릭하세요!", True, GRAY) screen.blit(guide, (WIDTH//2 - guide.get_width()//2, 450)) # --- 메인 실행 루프 --- running = True while running: clock.tick(FPS) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False # 0: 메인 메뉴 상태 이벤트 if game_state == 0: if event.type == pygame.MOUSEBUTTONDOWN: mx, my = event.pos if 250 <= mx <= 550 and 180 <= my <= 235: game_state = 1 # 타워 디펜스 실행 defense_game.reset() elif 250 <= mx <= 550 and 260 <= my <= 315: game_state = 2 # 삼매치 실행 elif 250 <= mx <= 550 and 340 <= my <= 395: game_state = 3 # 스마트폰 슈팅 실행 fps_game.reset() pygame.mouse.set_visible(False) # 게임 시작 시 마우스 숨김 # 1: 타워 디펜스 상태 이벤트 elif game_state == 1: if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: game_state = 0 # 메인 메뉴로 복귀 elif event.key == pygame.K_r and defense_game.life <= 0: defense_game.reset() elif event.type == pygame.MOUSEBUTTONDOWN: if defense_game.life > 0: defense_game.handle_click(event.pos) # 2: 삼매치 상태 이벤트 elif game_state == 2: if event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE: game_state = 0 # 3: 스마트폰 슈팅 게임 상태 이벤트 elif game_state == 3: if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: game_state = 0 pygame.mouse.set_visible(True) elif event.key == pygame.K_r: if fps_game.game_over: fps_game.reset() else: fps_game.reload() elif event.type == pygame.MOUSEBUTTONDOWN: if event.button == 1: if fps_game.game_over: fps_game.reset() else: fps_game.shoot(event.pos) # --- 화면 그리기 처리 --- if game_state == 0: draw_main_menu() elif game_state == 1: defense_game.update() defense_game.draw(screen) elif game_state == 2: # 추후 삼매치 오리지널 코딩 연동 자리 screen.fill(WHITE) msg = title_font.render("삼매치 게임 준비 중...", True, BLACK) sub_msg = font.render("삼매치 오리지널 코딩을 가져오시면 여기에 합쳐드립니다! [ESC: 메뉴]", True, GRAY) screen.blit(msg, (WIDTH//2 - msg.get_width()//2, 200)) screen.blit(sub_msg, (WIDTH//2 - sub_msg.get_width()//2, 280)) elif game_state == 3: fps_game.update() fps_game.draw(screen) pygame.display.flip() pygame.quit() sys.exit()
