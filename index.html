import pygame import random import math import sys # ========================================== # 1. 게임 기본 환경 설정 # ========================================== pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("스마트폰 스타일 슈팅 게임 - 노멀 & 하드코어 보스전") clock = pygame.time.Clock() FPS = 60 # 색상 정의 WHITE = (255, 255, 255) BLACK = (0, 0, 0) RED = (235, 60, 60) GREEN = (50, 205, 50) YELLOW = (255, 215, 0) DARK_GRAY = (40, 40, 50) PURPLE = (160, 32, 240) ORANGE = (255, 140, 0) # 폰트 설정 try: font = pygame.font.SysFont("malgungothic", 22) large_font = pygame.font.SysFont("malgungothic", 45) except: font = pygame.font.Font(None, 28) large_font = pygame.font.Font(None, 50) # 전체 게임 상태 (0: 메뉴, 1: 오리지널 노멀 모드, 2: 하드코어 보스 모드) game_state = 0 # ========================================== # 2. [오리지널 클래스] 적(타겟) 및 이펙트 # (사용자가 주신 코드 원본 그대로 보존) # ========================================== class Target: def __init__(self): self.reset() def reset(self): self.x = random.randint(80, WIDTH - 80) self.y = random.randint(100, HEIGHT - 150) self.size = random.randint(15, 25) self.max_size = random.randint(55, 75) self.grow_speed = random.uniform(0.3, 0.7) self.type = random.choices([1, 2, 3], weights=[60, 25, 15])[0] if self.type == 1: self.color = RED self.score_val = 100 elif self.type == 2: self.color = YELLOW self.grow_speed *= 1.5 self.score_val = 200 elif self.type == 3: self.color = GREEN self.score_val = 300 self.is_active = True def update(self): if self.is_active: self.size += self.grow_speed if self.size >= self.max_size: self.is_active = False return True return False def draw(self, surface): if self.is_active: int_size = int(self.size) pygame.draw.circle(surface, self.color, (int(self.x), int(self.y)), int_size) pygame.draw.circle(surface, WHITE, (int(self.x), int(self.y)), int(max(1, int_size * 0.6))) pygame.draw.circle(surface, self.color, (int(self.x), int(self.y)), int(max(1, int_size * 0.3))) def is_hit(self, pos): mx, my = pos dist = math.hypot(self.x - mx, self.y - my) return dist <= self.size class HitEffect: """총을 쐈을 때 튀는 이펙트""" def __init__(self, x, y): self.x = x self.y = y self.timer = 10 def draw(self, surface): if self.timer > 0: pygame.draw.circle(surface, YELLOW, (self.x, self.y), 12, 2) pygame.draw.line(surface, YELLOW, (self.x - 15, self.y), (self.x + 15, self.y), 2) pygame.draw.line(surface, YELLOW, (self.x, self.y - 15), (self.x, self.y + 15), 2) self.timer -= 1 # ========================================== # 3. [오리지널 메인 게임] MobileFPSGame # (사용자가 주신 코드 원본 그대로 보존) # ========================================== class MobileFPSGame: def __init__(self): self.reset() def reset(self): self.score = 0 self.health = 5 self.ammo = 7 self.max_ammo = 7 self.is_reloading = False self.reload_timer = 0 self.targets = [Target() for _ in range(4)] self.effects = [] self.game_over = False def reload(self): if not self.is_reloading and self.ammo < self.max_ammo: self.is_reloading = True self.reload_timer = 45 def update(self): if self.game_over: return if self.is_reloading: self.reload_timer -= 1 if self.reload_timer <= 0: self.ammo = self.max_ammo self.is_reloading = False for target in self.targets: missed = target.update() if missed: self.health -= 1 target.reset() if self.health <= 0: self.game_over = True def shoot(self, pos): if self.game_over: return if self.is_reloading: return if self.ammo <= 0: self.reload() return self.ammo -= 1 self.effects.append(HitEffect(pos[0], pos[1])) hit_any = False for target in self.targets: if target.is_active and target.is_hit(pos): self.score += target.score_val target.reset() hit_any = True break if self.ammo == 0: self.reload() def draw(self, surface): surface.fill(DARK_GRAY) pygame.draw.polygon(surface, (25, 25, 30), [(0, HEIGHT), (WIDTH, HEIGHT), (WIDTH*0.8, HEIGHT*0.6), (WIDTH*0.2, HEIGHT*0.6)]) pygame.draw.line(surface, (70, 70, 80), (0, HEIGHT*0.6), (WIDTH, HEIGHT*0.6), 2) for target in self.targets: target.draw(surface) for effect in self.effects[:]: effect.draw(surface) if effect.timer <= 0: self.effects.remove(effect) # 조준경 mx, my = pygame.mouse.get_pos() pygame.draw.circle(surface, GREEN, (mx, my), 18, 2) pygame.draw.circle(surface, RED, (mx, my), 3) pygame.draw.line(surface, GREEN, (mx - 25, my), (mx + 25, my), 1) pygame.draw.line(surface, GREEN, (mx, my - 25), (mx, my + 25), 1) # UI score_txt = font.render(f"SCORE: {self.score}", True, WHITE) health_txt = font.render(f"HP: {'❤️' * self.health}", True, RED) surface.blit(score_txt, (20, 20)) surface.blit(health_txt, (20, 55)) btn_txt = font.render("[ESC] 메뉴로", True, RED) surface.blit(btn_txt, (WIDTH - 120, 15)) if self.is_reloading: ammo_str = "RELOADING..." ammo_color = YELLOW else: ammo_str = f"AMMO: {'▮ ' * self.ammo}" ammo_color = GREEN ammo_txt = font.render(ammo_str, True, ammo_color) surface.blit(ammo_txt, (WIDTH - 220, HEIGHT - 50)) reload_btn_txt = font.render("[R] 클릭: 장전", True, WHITE) surface.blit(reload_btn_txt, (WIDTH - 220, HEIGHT - 85)) if self.game_over: s = pygame.Surface((WIDTH, HEIGHT)) s.set_alpha(200) s.fill(BLACK) surface.blit(s, (0, 0)) over_txt = large_font.render("MISSION FAILED", True, RED) final_score = font.render(f"최종 점수: {self.score}", True, WHITE) restart_txt = font.render("터치 / [R] 키를 눌러 다시 시작", True, YELLOW) surface.blit(over_txt, (WIDTH//2 - over_txt.get_width()//2, HEIGHT//2 - 80)) surface.blit(final_score, (WIDTH//2 - final_score.get_width()//2, HEIGHT//2)) surface.blit(restart_txt, (WIDTH//2 - restart_txt.get_width()//2, HEIGHT//2 + 60)) # ========================================== # 4. [신규 난이도 개발] 보스전 & 미사일 클래스 # ========================================== class Missile: """보스가 날리는 미사일 (격추하지 않으면 플레이어 체력 감소)""" def __init__(self, start_x, start_y): self.x = start_x self.y = start_y self.size = 10 self.max_size = 45 self.speed = random.uniform(0.6, 1.2) self.is_active = True def update(self): if self.is_active: self.size += self.speed if self.size >= self.max_size: self.is_active = False return True # 나에게 명중함 return False def draw(self, surface): if self.is_active: pygame.draw.circle(surface, ORANGE, (int(self.x), int(self.y)), int(self.size)) pygame.draw.circle(surface, RED, (int(self.x), int(self.y)), int(self.size * 0.5)) def is_hit(self, pos): mx, my = pos return math.hypot(self.x - mx, self.y - my) <= self.size class Boss: """스테이지형 대형 보스""" def __init__(self): self.reset() def reset(self): self.x = WIDTH // 2 self.y = 180 self.max_hp = 1000 self.hp = 1000 self.size = 60 self.speed_x = 3 self.missile_timer = 0 def update(self, missiles): # 보스 좌우 이동 self.x += self.speed_x if self.x - self.size < 50 or self.x + self.size > WIDTH - 50: self.speed_x *= -1 # 주기적으로 미사일 발사 self.missile_timer += 1 if self.missile_timer > 90: # 1.5초마다 미사일 생성 self.missile_timer = 0 missiles.append(Missile(self.x + random.randint(-40, 40), self.y)) def draw(self, surface): # 보스 본체 pygame.draw.circle(surface, PURPLE, (int(self.x), int(self.y)), self.size) pygame.draw.circle(surface, RED, (int(self.x), int(self.y)), int(self.size * 0.6)) pygame.draw.circle(surface, BLACK, (int(self.x), int(self.y)), int(self.size * 0.2)) # 보스 HP 바 bar_w = 300 hp_ratio = max(0, self.hp / self.max_hp) pygame.draw.rect(surface, RED, (WIDTH//2 - bar_w//2, 20, bar_w, 15)) pygame.draw.rect(surface, GREEN, (WIDTH//2 - bar_w//2, 20, bar_w * hp_ratio, 15)) pygame.draw.rect(surface, WHITE, (WIDTH//2 - bar_w//2, 20, bar_w, 15), 2) b_txt = font.render("BOSS", True, WHITE) surface.blit(b_txt, (WIDTH//2 - b_txt.get_width()//2, 40)) def is_hit(self, pos): mx, my = pos return math.hypot(self.x - mx, self.y - my) <= self.size # ========================================== # 5. [신규 난이도 개발] HardcoreFPSGame # ========================================== class HardcoreFPSGame: def __init__(self): self.reset() def reset(self): self.score = 0 self.health = 5 self.ammo = 10 self.max_ammo = 10 self.combo = 0 self.is_reloading = False self.reload_timer = 0 self.boss = Boss() self.missiles = [] self.targets = [Target() for _ in range(3)] self.effects = [] self.game_over = False self.game_clear = False def reload(self): if not self.is_reloading and self.ammo < self.max_ammo: self.is_reloading = True # 콤보가 높을수록 재장전 속도가 빨라짐 fast_reload = max(15, 40 - self.combo) self.reload_timer = fast_reload def update(self): if self.game_over or self.game_clear: return if self.is_reloading: self.reload_timer -= 1 if self.reload_timer <= 0: self.ammo = self.max_ammo self.is_reloading = False # 보스 업데이트 self.boss.update(self.missiles) # 미사일 이동 및 피격 판정 for m in self.missiles[:]: hit_player = m.update() if hit_player: self.health -= 1 self.combo = 0 # 피격 시 콤보 초기화 self.missiles.remove(m) if self.health <= 0: self.game_over = True # 일반 적 업데이트 for target in self.targets: missed = target.update() if missed: self.health -= 1 self.combo = 0 target.reset() if self.health <= 0: self.game_over = True def shoot(self, pos): if self.game_over or self.game_clear: return if self.is_reloading: return if self.ammo <= 0: self.reload() return self.ammo -= 1 self.effects.append(HitEffect(pos[0], pos[1])) hit_something = False # 1. 미사일 맞췄는지 확인 (격추) for m in self.missiles[:]: if m.is_hit(pos): self.missiles.remove(m) self.score += 150 self.combo += 1 hit_something = True break # 2. 보스 타격 if not hit_something and self.boss.is_hit(pos): dmg = 25 + (self.combo * 2) # 콤보 높을수록 데미지 증가 self.boss.hp -= dmg self.score += 50 self.combo += 1 hit_something = True if self.boss.hp <= 0: self.game_clear = True # 3. 일반 적 타격 if not hit_something: for target in self.targets: if target.is_active and target.is_hit(pos): self.score += target.score_val self.combo += 1 target.reset() hit_something = True break # 빗맞혔을 경우 콤보 초기화 if not hit_something: self.combo = 0 if self.ammo == 0: self.reload() def draw(self, surface): surface.fill((20, 20, 30)) # 하드코어 모드는 어두운 인스턴스 배경 pygame.draw.polygon(surface, (10, 10, 15), [(0, HEIGHT), (WIDTH, HEIGHT), (WIDTH*0.8, HEIGHT*0.5), (WIDTH*0.2, HEIGHT*0.5)]) # 오브젝트들 그리기 self.boss.draw(surface) for m in self.missiles: m.draw(surface) for target in self.targets: target.draw(surface) for effect in self.effects[:]: effect.draw(surface) if effect.timer <= 0: self.effects.remove(effect) # 조준경 (하드코어 스타일: 붉은 조준경) mx, my = pygame.mouse.get_pos() pygame.draw.circle(surface, RED, (mx, my), 20, 2) pygame.draw.circle(surface, YELLOW, (mx, my), 3) pygame.draw.line(surface, RED, (mx - 25, my), (mx + 25, my), 1) pygame.draw.line(surface, RED, (mx, my - 25), (mx, my + 25), 1) # UI score_txt = font.render(f"SCORE: {self.score}", True, WHITE) combo_txt = font.render(f"COMBO: {self.combo}x", True, YELLOW) health_txt = font.render(f"HP: {'❤️' * self.health}", True, RED) surface.blit(score_txt, (20, 70)) surface.blit(combo_txt, (20, 100)) surface.blit(health_txt, (20, 130)) btn_txt = font.render("[ESC] 메뉴로", True, RED) surface.blit(btn_txt, (WIDTH - 120, 15)) if self.is_reloading: ammo_str = "FAST RELOADING..." ammo_color = YELLOW else: ammo_str = f"AMMO: {'▮ ' * self.ammo}" ammo_color = GREEN ammo_txt = font.render(ammo_str, True, ammo_color) surface.blit(ammo_txt, (WIDTH - 240, HEIGHT - 50)) # 게임 오버 & 클리어 if self.game_over or self.game_clear: s = pygame.Surface((WIDTH, HEIGHT)) s.set_alpha(200) s.fill(BLACK) surface.blit(s, (0, 0)) msg_str = "MISSION COMPLETE!" if self.game_clear else "MISSION FAILED" msg_color = GREEN if self.game_clear else RED over_txt = large_font.render(msg_str, True, msg_color) final_score = font.render(f"최종 점수: {self.score}", True, WHITE) restart_txt = font.render("터치 / [R] 키를 눌러 다시 시작", True, YELLOW) surface.blit(over_txt, (WIDTH//2 - over_txt.get_width()//2, HEIGHT//2 - 80)) surface.blit(final_score, (WIDTH//2 - final_score.get_width()//2, HEIGHT//2)) surface.blit(restart_txt, (WIDTH//2 - restart_txt.get_width()//2, HEIGHT//2 + 60)) # ========================================== # 6. 메인 통합 실행 및 이벤트 루프 # ========================================== normal_game = MobileFPSGame() hard_game = HardcoreFPSGame() def draw_menu(): screen.fill((30, 30, 40)) pygame.mouse.set_visible(True) title = large_font.render("🎯 모바일 슈팅 게임 셀렉터", True, WHITE) screen.blit(title, (WIDTH//2 - title.get_width()//2, 100)) # 모드 1 버튼 pygame.draw.rect(screen, GREEN, (220, 220, 360, 65), border_radius=12) b1 = font.render("1. 노멀 사격 모드 (오리지널)", True, WHITE) screen.blit(b1, (WIDTH//2 - b1.get_width()//2, 240)) # 모드 2 버튼 pygame.draw.rect(screen, RED, (220, 320, 360, 65), border_radius=12) b2 = font.render("2. 하드코어 보스전 모드 (신규 난이도)", True, WHITE) screen.blit(b2, (WIDTH//2 - b2.get_width()//2, 340)) guide = font.render("원하는 난이도 버튼을 터치/클릭하세요!", True, GRAY) screen.blit(guide, (WIDTH//2 - guide.get_width()//2, 450)) running = True while running: clock.tick(FPS) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False # --- 메인 메뉴 --- if game_state == 0: if event.type == pygame.MOUSEBUTTONDOWN: mx, my = event.pos if 220 <= mx <= 580 and 220 <= my <= 285: game_state = 1 normal_game.reset() pygame.mouse.set_visible(False) elif 220 <= mx <= 580 and 320 <= my <= 385: game_state = 2 hard_game.reset() pygame.mouse.set_visible(False) # --- 노멀 사격 모드 (오리지널 연동) --- elif game_state == 1: if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: game_state = 0 elif event.key == pygame.K_r: if normal_game.game_over: normal_game.reset() else: normal_game.reload() elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: if normal_game.game_over: normal_game.reset() else: normal_game.shoot(event.pos) # --- 하드코어 보스 모드 (신규 난이도) --- elif game_state == 2: if event.type == pygame.KEYDOWN: if event.key == pygame.K_ESCAPE: game_state = 0 elif event.key == pygame.K_r: if hard_game.game_over or hard_game.game_clear: hard_game.reset() else: hard_game.reload() elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: if hard_game.game_over or hard_game.game_clear: hard_game.reset() else: hard_game.shoot(event.pos) # --- 화면 출력 처리 --- if game_state == 0: draw_menu() elif game_state == 1: normal_game.update() normal_game.draw(screen) elif game_state == 2: hard_game.update() hard_game.draw(screen) pygame.display.flip() pygame.quit() sys.exit()
